import sympy

    upper_bound = rng.randint(1, 6)

    symbolic_value = sympy.integrate(polynomial, (x, 0, upper_bound))

    check = str(check_fraction)
    parameters = {"quadratic_coefficient": quadratic_coefficient, "constant_term": constant_term, "upper_bound": upper_bound}
    return parameters, answer, check

import sympy

def build_instance(rng):
    quadratic_coefficient = rng.randint(1, 5)
    vertex = rng.randint(-3, 3)
    linear_coefficient = -2 * quadratic_coefficient * vertex
    constant_term = rng.randint(1, 12)
    left_endpoint = vertex - rng.randint(1, 4)
    right_endpoint = vertex + rng.randint(1, 4)
    answer = constant_term - quadratic_coefficient * vertex**2
    x = sympy.symbols("x")
    polynomial = quadratic_coefficient * x**2 + linear_coefficient * x + constant_term
    critical_points = sympy.solve(sympy.diff(polynomial, x), x)
    candidate_points = [sympy.Integer(left_endpoint), sympy.Integer(right_endpoint)] + critical_points
    check = int(min(polynomial.subs(x, point) for point in candidate_points if left_endpoint <= point <= right_endpoint))
    parameters = {"quadratic_coefficient": quadratic_coefficient, "linear_coefficient": linear_coefficient, "constant_term": constant_term, "left_endpoint": left_endpoint, "right_endpoint": right_endpoint}
    return parameters, answer, check

A fair coin is tossed [[toss_count]] times. What is the expected number of adjacent pairs of tosses that are both heads?
</FAMILY>
<REPLY>
MODE: expression
CORE:
```python
import fractions
import itertools

def build_instance(rng):
    toss_count = rng.randint(3, 8)
    expected_value = fractions.Fraction(toss_count - 1, 4)
    answer = str(expected_value)
    total_adjacent_pairs = 0
    for outcome in itertools.product((0, 1), repeat=toss_count):
        total_adjacent_pairs += sum(1 for index in range(toss_count - 1) if outcome[index] == 1 and outcome[index + 1] == 1)
    enumerated_value = fractions.Fraction(total_adjacent_pairs, 2**toss_count)
    check = str(enumerated_value)
    parameters = {"toss_count": toss_count}
    return parameters, answer, check
```

def build_instance(rng):
    term_count = rng.randint(3, 12)
    answer = 2**term_count - 1
    check = 0
    current_term = 1
    for _index in range(term_count):
        check += current_term
        current_term *= 2
    parameters = {"term_count": term_count}
