NOW WRITE THE GENERATOR FOR THIS FAMILY

The numbers 1 through {board_size} are written on a board. In each move, two numbers a and b are erased and the single number a + b - 1 is written in their place. After {board_size} - 1 moves one number remains. Determine that number.

CORRECT OUTPUT:
```python
import random

def generate(seed):
    rng = random.Random(seed)
    board_size = rng.randint(5, 12)

    # Each move removes two numbers and writes one, so the count drops by 1 and
    # the sum drops by 1: sum - count never changes.
    answer = board_size * (board_size + 1) // 2 - board_size + 1

    # Independent route: actually play the game in a random order.
    pool = list(range(1, board_size + 1))
    while len(pool) > 1:
        a = pool.pop(rng.randrange(len(pool)))
        b = pool.pop(rng.randrange(len(pool)))
        pool.append(a + b - 1)
    check = pool[0]
    assert answer == check, f"answer={answer} check={check}"

    problem = f"The numbers 1 through {board_size} are written on a board. In each move, two numbers a and b are erased and the single number a + b - 1 is written in their place. After {board_size} - 1 moves one number remains. Determine that number."
    return problem, str(answer)
```

import random

def generate(seed):

    first_term = rng.randint(1, 20)
    common_difference = rng.randint(1, 9)
    term_index = rng.randint(3, 15)

    answer = first_term + (term_index - 1) * common_difference
    check = first_term

    assert answer == check, f"answer={answer} check={check}"

    problem = f"Let a = {first_term} and d = {common_difference}. What is the {term_index}-th term of the arithmetic progression starting at a with common difference d?"

NOW WRITE THE GENERATOR FOR THIS FAMILY

A sequence of positive integers satisfies a_1 = {start_value} and a_(k+1) = 3 * a_k + {shift} for every k >= 1. Find a_{index}.

CORRECT OUTPUT:
```python
import random

def generate(seed):
    rng = random.Random(seed)
    start_value = rng.randint(1, 9)
    shift = rng.randint(1, 9)
    index = rng.randint(4, 9)

    # Unrolling the recurrence once gives 3^(k-1) * start + shift * (3^(k-1) - 1) / 2.
    power = 3 ** (index - 1)
    answer = power * start_value + shift * (power - 1) // 2

    # Independent route: iterate the recurrence itself, one step at a time.
    check = start_value
    for _ in range(index - 1):
        check = 3 * check + shift
    assert answer == check, f"answer={answer} check={check}"

    problem = f"A sequence of positive integers satisfies a_1 = {start_value} and a_(k+1) = 3 * a_k + {shift} for every k >= 1. Find a_{index}."
    return problem, str(answer)
```

NOW WRITE THE GENERATOR FOR THIS FAMILY

Let N = {upper_bound}. A set S of integers from 1 to N has the property that no element of S divides another. What is the largest possible number of elements of S?

CORRECT OUTPUT:
```python
import random

def generate(seed):
    rng = random.Random(seed)
    upper_bound = rng.randint(8, 13)

    # Any ceil(N/2)+1 of the numbers contain two with the same odd part, and
    # then the smaller divides the larger -- so no larger set can exist.
    answer = (upper_bound + 1) // 2

    # Independent route: search every subset for the largest divisor-free one.
    check = 0
    for mask in range(1 << upper_bound):
        chosen = [k + 1 for k in range(upper_bound) if mask >> k & 1]
        if len(chosen) <= check:
            continue
        if all(b % a for i, a in enumerate(chosen) for b in chosen[i + 1:]):
            check = len(chosen)
    assert answer == check, f"answer={answer} check={check}"

    problem = f"Let N = {upper_bound}. A set S of integers from 1 to N has the property that no element of S divides another. What is the largest possible number of elements of S?"
    return problem, str(answer)
```

EXAMPLE -- this family

Let a = {modulus} and b = {lower_bound} be positive integers. What is the smallest multiple of a that is strictly greater than b?

is produced by this generator:
```python
import random

def generate(seed):
    rng = random.Random(seed)
    modulus = rng.randint(2, 15)
    lower_bound = rng.randint(1, 100)

    answer = (lower_bound // modulus + 1) * modulus
    x = lower_bound + 1
    while x % modulus:
        x += 1
    check = x
    assert answer == check, f"answer={answer} check={check}"

    problem = f"Let a = {modulus} and b = {lower_bound} be positive integers. What is the smallest multiple of a that is strictly greater than b?"
    return problem, str(answer)
```

NOW WRITE THE GENERATOR FOR THIS FAMILY

Find the positive integer with exactly {digit_count} digits, every digit equal to 1 or 2, that is divisible by 2^{digit_count}.

CORRECT OUTPUT:
```python
import random

def generate(seed):
    rng = random.Random(seed)
    digit_count = rng.randint(4, 11)

    # Build the number one digit at a time from the right: at step k exactly one
    # of the two digits keeps the k-digit suffix divisible by 2^k.
    answer = 0
    for k in range(digit_count):
        for digit in (1, 2):
            candidate = digit * 10 ** k + answer
            if candidate % 2 ** (k + 1) == 0:
                answer = candidate
                break

    # Independent route: try every string of 1s and 2s of that length.
    check = None
    for mask in range(1 << digit_count):
        value = 0
        for k in range(digit_count):
            value += (2 if mask >> k & 1 else 1) * 10 ** k
        if value % 2 ** digit_count == 0:
            check = value if check is None else min(check, value)
    assert answer == check, f"answer={answer} check={check}"

    problem = f"Find the positive integer with exactly {digit_count} digits, every digit equal to 1 or 2, that is divisible by 2^{digit_count}."
    return problem, str(answer)
```
