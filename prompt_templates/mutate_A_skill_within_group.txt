EXAMPLE_PARENT:
```python
import random

GROUP = "geometry"
SKILL = "transformation"


def generate(seed):
    rng = random.Random(seed)
    # B sits horizontally from A and C is sheared sideways, so the cross
    # product is p*q for every draw and the triangle can never degenerate.
    ax, ay = rng.randint(-9, 9), rng.randint(-9, 9)
    p, q = rng.randint(6, 14), rng.randint(6, 14)
    t = rng.randint(-10, 10)
    bx, by = ax + p, ay
    cx, cy = ax + t, ay + q

    answer = abs((bx - ax) * (cy - ay) - (cx - ax) * (by - ay))

    # Independent route: the shoelace sum around the whole triangle, instead of
    # one cross product of the two edges leaving A.
    pts = [(ax, ay), (bx, by), (cx, cy)]
    check = abs(sum(pts[i][0] * pts[(i + 1) % 3][1] - pts[(i + 1) % 3][0] * pts[i][1]
                    for i in range(3)))
    assert answer == check, f"answer={answer} check={check}"

    problem = (
        f"Triangle ABC has vertices A = ({ax}, {ay}), B = ({bx}, {by}) and "
        f"C = ({cx}, {cy}). Find twice its area. State only the integer."
    )
    return problem, str(answer)
```

EXAMPLE_CHILD:
SKILL_RATIONALE: The parent's transformation move hands a solver the shoelace formula, which returns the area; area alone cannot decide how many lattice points lie strictly inside, because two triangles of equal area can contain different numbers of them, so what is left is to count them. -> counting
```python
import math
import random

GROUP = "geometry"
SKILL = "counting"


def generate(seed):
    rng = random.Random(seed)
    ax, ay = rng.randint(-9, 9), rng.randint(-9, 9)
    p, q = rng.randint(6, 14), rng.randint(6, 14)
    t = rng.randint(-10, 10)
    bx, by = ax + p, ay
    cx, cy = ax + t, ay + q

    # Pick's theorem: interior = area - boundary/2 + 1, with the boundary count
    # of each edge given by the gcd of its coordinate differences.
    twice_area = abs((bx - ax) * (cy - ay) - (cx - ax) * (by - ay))
    boundary = (math.gcd(abs(bx - ax), abs(by - ay))
                + math.gcd(abs(cx - bx), abs(cy - by))
                + math.gcd(abs(ax - cx), abs(ay - cy)))
    answer = (twice_area - boundary + 2) // 2

    # Independent route: test every lattice point of the bounding box against
    # the three edges, instead of combining area and boundary through Pick.
    def side(px, py, qx, qy, rx, ry):
        return (qx - px) * (ry - py) - (rx - px) * (qy - py)

    check = 0
    for x in range(min(ax, bx, cx), max(ax, bx, cx) + 1):
        for y in range(min(ay, by, cy), max(ay, by, cy) + 1):
            s1 = side(ax, ay, bx, by, x, y)
            s2 = side(bx, by, cx, cy, x, y)
            s3 = side(cx, cy, ax, ay, x, y)
            if (s1 > 0 and s2 > 0 and s3 > 0) or (s1 < 0 and s2 < 0 and s3 < 0):
                check += 1
    assert answer == check, f"answer={answer} check={check}"

    problem = (
        f"Triangle ABC has vertices A = ({ax}, {ay}), B = ({bx}, {by}) and "
        f"C = ({cx}, {cy}). How many points with integer coordinates lie "
        f"strictly inside it? State only the integer."
    )
    return problem, str(answer)
```
