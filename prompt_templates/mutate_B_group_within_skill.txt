EXAMPLE_PARENT:
```python
import math
import random

GROUP = "geometry"
SKILL = "counting"


def generate(seed):
    rng = random.Random(seed)
    w, h = rng.randint(12, 30), rng.randint(12, 30)

    # Complement over the grid, with the union of the two diagonals expanded by
    # inclusion-exclusion. A segment from (0,0) to (w,h) meets gcd(w,h)+1
    # lattice points, and the diagonals cross at the centre, which is a lattice
    # point exactly when w and h are both even.
    total = (w + 1) * (h + 1)
    on_each = math.gcd(w, h) + 1
    on_both = 1 if w % 2 == 0 and h % 2 == 0 else 0
    answer = total - (2 * on_each - on_both)

    # Independent route: walk every lattice point of the rectangle and test it
    # against the two diagonal equations, instead of counting by complement.
    check = sum(
        1
        for x in range(w + 1)
        for y in range(h + 1)
        if x * h != y * w and x * h + y * w != w * h
    )
    assert answer == check, f"answer={answer} check={check}"

    problem = (
        f"Consider the grid of points with integer coordinates 0 <= x <= {w} "
        f"and 0 <= y <= {h}. How many of them lie on neither diagonal of the "
        f"rectangle? State only the integer."
    )
    return problem, str(answer)
```

GROUP_RATIONALE: The objects moved from diagonals of a polygon to integers divisible by given primes, and solving still takes counting: the answer is a total minus a union that has to be expanded by inclusion-exclusion, exactly the move the parent used on a different kind of object.
```python
import random

GROUP = "number_theory"
SKILL = "counting"


def generate(seed):
    rng = random.Random(seed)
    primes = sorted(rng.sample([3, 5, 7, 11, 13, 17], 3))
    n = rng.randint(3000, 40000)

    # Same shape as the parent: total minus the union, with the union expanded
    # by inclusion-exclusion. Divisibility replaces incidence as the relation
    # that decides membership.
    p, q, r = primes
    union = (n // p + n // q + n // r
             - n // (p * q) - n // (p * r) - n // (q * r)
             + n // (p * q * r))
    answer = n - union

    # Independent route: test every integer in range against the three stated
    # divisors, instead of counting by complement.
    check = sum(1 for k in range(1, n + 1) if k % p and k % q and k % r)
    assert answer == check, f"answer={answer} check={check}"

    problem = (
        f"How many integers k with 1 <= k <= {n} are divisible by none of "
        f"{p}, {q} and {r}? State only the integer."
    )
    return problem, str(answer)
```
