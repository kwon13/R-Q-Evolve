import random
import sympy

def generate(seed):
    rng = random.Random(seed)
    n = rng.randint(300, 3000)
    factors = sympy.factorint(n)
    answer = 1
    for exponent in sorted(factors.values()):
        answer *= exponent + 1
    return f"How many positive divisors does {n} have?", str(sympy.Integer(answer))
