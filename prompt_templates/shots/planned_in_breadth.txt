"""Metacognitive in-breadth transfer from divisor counting to grid boxes.

TARGET REASONING MOVE
    Decompose into independent components and multiply counts after deriving
    one extra boundary choice per component.

WHAT CHANGES
    Integer factor exponents become three unit-cell axes; selecting exponent
    values becomes selecting pairs of boundary planes. The vocabulary and
    CONCEPT_GROUP change from number theory to combinatorics.

ANSWER ROUTE
    Compute C(a+1,2)C(b+1,2)C(c+1,2).
"""

import random
import sympy

MAX_ATTEMPTS = 200


def generate(seed):
    rng = random.Random(seed)

    for _ in range(MAX_ATTEMPTS):
        a = rng.randint(3, 7)
        b = rng.randint(4, 8)
        c = rng.randint(3, 7)

        answer = int(
            sympy.binomial(a + 1, 2)
            * sympy.binomial(b + 1, 2)
            * sympy.binomial(c + 1, 2)
        )
        if answer < 100:
            continue
        break
    else:
        raise RuntimeError(
            f"generator exhausted MAX_ATTEMPTS={MAX_ATTEMPTS} for seed={seed}"
        )

    problem = (
        f"A rectangular solid is divided into a {a} by {b} by {c} grid of "
        f"unit cubes. How many axis-aligned rectangular boxes have all vertices "
        f"on grid points? State only the integer count."
    )
    return problem, str(sympy.Integer(answer))


CONCEPT_REASON = "The solver must decompose the box choice by axis and recognize that an axis with a unit cells has a+1 boundary planes, then multiply the three boundary-pair counts."
CONCEPT_GROUP = "combinatorics"
CONCEPT_TYPE = "combinatorics.grid_rectangular_boxes"
